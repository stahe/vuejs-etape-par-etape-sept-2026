    // =========================================================================
// vite.config.ts - la configuration de Vite, commune à TOUS les exemples.
// -------------------------------------------------------------------------
// Vite est à la fois :
//   - un serveur de développement (npm run dev) : il compile à la volée les
//     fichiers .vue et .ts demandés par le navigateur, et recharge la page
//     (ou seulement le composant modifié) à chaque enregistrement ;
//   - un outil de construction (npm run build) : il produit, dans dist/,
//     des fichiers .js et .css optimisés à déposer sur un serveur web.
//
// Chaque exemple est une petite application indépendante, avec son propre
// index.html : 01-premiere-application/index.html, 02-reactivite/index.html...
// Un seul serveur Vite les sert toutes : http://localhost:5173/ affiche le
// sommaire (index.html de ce dossier), http://localhost:5173/01-premiere-application/
// le premier exemple, etc.
// =========================================================================

import { existsSync, readdirSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { resolve } from 'node:path';
import { defineConfig } from 'vite';
import vue from '@vitejs/plugin-vue';

const racine = fileURLToPath(new URL('.', import.meta.url));

// les dossiers d'exemples : ceux dont le nom commence par deux chiffres
const exemples = readdirSync(racine).filter((d) => /^\d\d-/.test(d) && existsSync(resolve(racine, d, 'index.html')));

export default defineConfig({
  // le plugin qui apprend à Vite à compiler les fichiers .vue
  plugins: [vue()],

  server: {
    port: 5173,
    // le PROXY : toute requête du navigateur vers /api/... est transmise par
    // Vite au serveur JSON RdvMedecins (port 8080), puis la réponse revient
    // au navigateur. Pour le navigateur, page et API ont donc la MÊME origine
    // (http://localhost:5173) : pas de CORS à configurer, et le cookie du
    // jeton JWT circule normalement (cf. chapitre sur la boîte noire).
    proxy: {
      '/api': 'http://localhost:8080',
    },
  },

  // npm run build : une page HTML d'entrée par exemple, plus le sommaire
  build: {
    rollupOptions: {
      input: Object.fromEntries([
        ['sommaire', resolve(racine, 'index.html')],
        ...exemples.map((d) => [d, resolve(racine, d, 'index.html')]),
      ]),
    },
  },
});
    • ligne 31 : l'extension qui apprend à [Vite] à compiler les fichiers [.vue]. Un navigateur ne sait pas lire un fichier [.vue] : quand il en demande un, [Vite] le transforme à la volée en [JavaScript] ordinaire ;
    • ligne 34 : le port du serveur de développement ;
    • lignes 40-41 : le proxy, qui ne servira qu'à partir de l'exemple 17 : toute requête du navigateur vers une adresse commençant par [/api] est transmise au serveur JSON [RdvMedecins], sur le port 8080. Nous y reviendrons ;
    • ligne 27 et lignes 46-53 : [npm run build] doit connaître toutes les pages d'entrée de l'application. On lui donne le sommaire et le fichier [index.html] de chaque dossier dont le nom commence par deux chiffres.

    <!--
  =========================================================================
  01-premiere-application/src/App.vue - un premier composant
  -------------------------------------------------------------------------
  Un fichier .vue (Single File Component) regroupe les trois facettes d'un
  composant :
    - <script setup> : les données et le code du composant (TypeScript) ;
    - <template>     : le HTML à afficher, avec des {{ ... }} ;
    - <style>        : son CSS (ici "scoped" : il ne s'applique qu'à lui).
  =========================================================================
-->
<script setup lang="ts">
import { ref } from 'vue';

// des données "ordinaires" : le template peut les afficher
const titre = 'Bienvenue dans Vue.js';
const langages = ['TypeScript', 'HTML', 'CSS'];

// ref(...) : une donnée RÉACTIVE. Quand sa valeur (.value) change, Vue met
// à jour AUTOMATIQUEMENT les endroits de la page qui l'affichent
const compteur = ref(0);
const heure = ref(new Date().toLocaleTimeString());

// une fonction appelée par le bouton du template
function incrementer(): void {
  compteur.value++;
  heure.value = new Date().toLocaleTimeString();
}
</script>

<template>
  <div class="container py-4">
    <!-- {{ expression }} : la valeur de l'expression, insérée dans la page -->
    <h1 class="h3">{{ titre }}</h1>
    <p>Cette page est fabriquée dans le navigateur, par Vue.js, à partir de trois fichiers : index.html, main.ts et App.vue.</p>
    <p>On y écrit du {{ langages.join(', ') }} ({{ langages.length }} langages).</p>

    <!-- @click : la fonction à appeler quand on clique (cf. exemple 04) -->
    <button class="btn btn-primary" @click="incrementer">Cliquez-moi</button>
    <!-- dans le template, pas de .value : Vue "déballe" les ref tout seul -->
    <p class="mt-3 compteur">Vous avez cliqué {{ compteur }} fois (dernier clic : {{ heure }}).</p>
  </div>
</template>

<style scoped>
/* "scoped" : cette règle ne s'applique qu'aux éléments de CE composant */
.compteur {
  font-weight: bold;
  color: #0d6efd;
}
</style>

    Un fichier [.vue] (Single File Component, ou SFC) regroupe les trois facettes d'un composant :

    • lignes 12-29 : le code du composant, en [TypeScript] ([lang="ts"]). Avec [setup], toutes les variables et fonctions déclarées au premier niveau du script sont directement utilisables dans le template ;
      • lignes 16-17 : deux données ordinaires ;
      • ligne 21 : [ref(0)] crée une donnée réactive, de valeur initiale 0. On lit et on modifie sa valeur par sa propriété [value]. Quand elle change, [Vue.js] met à jour tous les endroits de la page qui l'affichent ;
      • lignes 25-27 : une fonction qui modifie deux données réactives. Elle ne touche pas à la page : c'est [Vue.js] qui la mettra à jour ;
    • lignes 31-43 : le HTML du composant :
      • ligne 34 : la syntaxe [{{ expression }}] (les « moustaches ») insère dans la page la valeur d'une expression [TypeScript]. Elle est échappée : si [titre] contenait [<b>], on verrait [<b>] s'afficher, sans que la balise soit interprétée ;
      • ligne 36 : l'expression peut être quelconque : un appel de méthode, un calcul ;
      • ligne 39 : un clic sur le bouton appelle la fonction [incrementer] (cf. exemple 04) ;
      • ligne 41 : dans le template, on écrit [compteur] et non [compteur.value] : [Vue.js] « déballe » automatiquement les [ref] ;
    • lignes 45-51 : le CSS du composant. Avec [scoped], ses règles ne s'appliquent qu'aux éléments de CE composant : [Vite] ajoute un attribut unique à ses éléments ([data-v-xxxx]) et le reporte dans les sélecteurs.

    <!--
  =========================================================================
  02-reactivite/src/App.vue - ref, reactive, computed, watch
  -------------------------------------------------------------------------
  Un petit calculateur de prix TTC :
    - ref      : une valeur réactive simple (un nombre, une chaîne...) ;
    - reactive : un OBJET réactif (toutes ses propriétés le sont) ;
    - computed : une valeur CALCULÉE à partir d'autres, recalculée
                 automatiquement quand elles changent ;
    - watch    : du code exécuté quand une valeur change (un "effet").
  =========================================================================
-->
<script setup lang="ts">
import { computed, reactive, ref, watch } from 'vue';

// ref : on lit et on écrit sa valeur par .value (dans le code TypeScript)
const quantite = ref(1);

// reactive : un objet, dont on lit et écrit les propriétés directement
const article = reactive({
  libelle: 'Stylo',
  prixHT: 2.5,
  tauxTVA: 20, // en %
});

// computed : jamais affecté directement, toujours recalculé. Vue sait de
// quelles données il dépend (quantite, article.prixHT, article.tauxTVA) :
// il le recalcule quand l'une d'elles change, et seulement dans ce cas
const totalHT = computed(() => quantite.value * article.prixHT);
const montantTVA = computed(() => (totalHT.value * article.tauxTVA) / 100);
const totalTTC = computed(() => totalHT.value + montantTVA.value);

// un formateur de montants en euros, à la française (1 234,50 €)
const euros = new Intl.NumberFormat('fr-FR', { style: 'currency', currency: 'EUR' });

// watch : à chaque changement du total TTC, on garde une trace
const historique = ref<string[]>([]);
watch(totalTTC, (nouveau, ancien) => {
  historique.value.unshift(`${euros.format(ancien)} -> ${euros.format(nouveau)}`);
  historique.value = historique.value.slice(0, 5); // les 5 derniers
});

function reinitialiser(): void {
  quantite.value = 1;
  article.prixHT = 2.5;
  article.tauxTVA = 20;
}
</script>

<template>
  <div class="container py-4">
    <h1 class="h3 mb-3">Exemple 02 - la réactivité</h1>
    <div class="row g-4">
      <div class="col-md-6">
        <div class="card p-3">
          <!-- v-model relie un champ à une donnée, dans les deux sens (cf. exemple 05) -->
          <label class="form-label">Article</label>
          <input v-model="article.libelle" class="form-control mb-2">
          <label class="form-label">Prix unitaire HT</label>
          <input v-model.number="article.prixHT" type="number" step="0.1" min="0" class="form-control mb-2">
          <label class="form-label">Quantité</label>
          <input v-model.number="quantite" type="number" min="0" class="form-control mb-2">
          <label class="form-label">Taux de TVA</label>
          <select v-model.number="article.tauxTVA" class="form-select mb-3">
            <option :value="5.5">5,5 %</option>
            <option :value="10">10 %</option>
            <option :value="20">20 %</option>
          </select>
          <button class="btn btn-outline-secondary" @click="reinitialiser">Réinitialiser</button>
        </div>
      </div>
      <div class="col-md-6">
        <div class="card p-3">
          <h2 class="h5">{{ quantite }} x {{ article.libelle }}</h2>
          <table class="table mb-0">
            <tbody>
              <tr><td>Total HT</td><td class="text-end">{{ euros.format(totalHT) }}</td></tr>
              <tr><td>TVA ({{ article.tauxTVA }} %)</td><td class="text-end">{{ euros.format(montantTVA) }}</td></tr>
              <tr class="fw-bold"><td>Total TTC</td><td class="text-end">{{ euros.format(totalTTC) }}</td></tr>
            </tbody>
          </table>
        </div>
        <div class="card p-3 mt-3">
          <h2 class="h6">Historique du total TTC (watch)</h2>
          <ul class="mb-0">
            <li v-for="(ligne, i) in historique" :key="i">{{ ligne }}</li>
          </ul>
        </div>
      </div>
    </div>
  </div>
</template>
    • ligne 17 : [ref] convient à une valeur simple (nombre, chaîne, booléen) ;
    • lignes 20-24 : [reactive] rend réactif un objet entier : chacune de ses propriétés l'est. On les lit et on les écrit directement ([article.prixHT]), sans [.value] ;
    • lignes 29-31 : [computed] définit une valeur calculée à partir d'autres. [Vue.js] note les données lues pendant le calcul ([quantite], [article.prixHT]...) et ne refait le calcul que lorsque l'une d'elles change. On ne l'affecte jamais : on la lit ([totalHT.value]) ;
    • ligne 34 : [Intl.NumberFormat], l'outil de mise en forme des nombres intégré à [JavaScript] : [2.5] devient « 2,50 € » ;
    • lignes 38-41 : [watch] exécute du code chaque fois qu'une donnée change, en recevant sa nouvelle et son ancienne valeur. On l'utilise pour les effets : mémoriser, envoyer une requête, écrire dans le stockage du navigateur... Pour afficher une valeur dérivée, on préfère [computed] ;
    • lignes 43-46 : modifier les données suffit : tous les champs, totaux et l'historique suivent ;
    • lignes 58-64 : [v-model] relie chaque champ à une donnée, dans les deux sens (cf. exemple 05) ;
    • ligne 77 : dans le template, un [computed] s'utilise comme une [ref], sans [.value] ;
    • ligne 86 : une liste répétée pour chaque élément du tableau [historique] (cf. exemple 03).

  =========================================================================
  03-directives/src/App.vue - les directives de base
  -------------------------------------------------------------------------
  Une DIRECTIVE est un attribut spécial du template, préfixé par v- :
    - v-bind:attr="expr" (abrégé :attr="expr") : la valeur d'un attribut
      HTML calculée par une expression ; :class et :style acceptent des

// une interface TypeScript décrit la forme des objets de la liste
interface Medecin {

  prenom: string;

  specialite: string;
  disponible: boolean;

    (m) =>
      (!seulementDisponibles.value || m.disponible) &&

    <p v-if="medecins.length === 0" class="text-muted">Aucun médecin.</p>
    <p v-else-if="medecinsAffiches.length === 0" class="text-warning">Aucun médecin ne correspond à « {{ filtre }} ».</p>

<script setup lang="ts">
import { ref } from 'vue';

interface Tache {
  id: number;
  libelle: string;
  faite: boolean;
}

const taches = ref<Tache[]>([
  { id: 1, libelle: 'Installer Node.js', faite: true },
  { id: 2, libelle: 'Lire le chapitre sur les événements', faite: false },
]);
const nouvelle = ref('');
const journal = ref<string[]>([]);
let prochainId = 3;

function tracer(message: string): void {
  journal.value.unshift(`${new Date().toLocaleTimeString()} ${message}`);
}

// appelée par @submit.prevent : le navigateur n'envoie PAS le formulaire au
// serveur (pas de rechargement de la page), c'est notre code qui agit
function ajouter(): void {
  const libelle = nouvelle.value.trim();
  if (libelle === '') {
    return;
  }
  taches.value.push({ id: prochainId++, libelle, faite: false });
  tracer(`ajout de « ${libelle} »`);
  nouvelle.value = '';
}

function supprimer(tache: Tache): void {
  taches.value = taches.value.filter((t) => t.id !== tache.id);
  tracer(`suppression de « ${tache.libelle} »`);
}

// $event : l'événement natif (ici un MouseEvent), avec ses coordonnées
function position(evenement: MouseEvent): void {
  tracer(`clic en (${evenement.offsetX}, ${evenement.offsetY})`);
}
</script>

<template>
  <div class="container py-4">

      <input v-model="nouvelle" class="form-control" placeholder="nouvelle tâche" @keyup.esc="nouvelle = ''">
      <button type="submit" class="btn btn-primary">Ajouter</button>
    </form>

    <ul class="list-group mb-3">
      <li v-for="t in taches" :key="t.id" class="list-group-item d-flex align-items-center gap-2">
        <input v-model="t.faite" type="checkbox" class="form-check-input" @change="tracer(`« ${t.libelle} » : ${t.faite ? 'faite' : 'à faire'}`)">
        <span :class="{ 'text-decoration-line-through text-muted': t.faite }" class="flex-grow-1">{{ t.libelle }}</span>
        <!-- un gestionnaire avec argument : la tâche de CETTE ligne -->
        <button class="btn btn-sm btn-outline-danger" @click="supprimer(t)">Supprimer</button>
      </li>
    </ul>

    <div class="zone p-3 mb-3 rounded" @click="position($event)">Cliquez dans cette zone</div>

    <h2 class="h6">Journal</h2>

    <!--
  =========================================================================
  06-validation/src/App.vue - valider un formulaire dans le navigateur
  -------------------------------------------------------------------------
  Les erreurs sont des COMPUTED : elles sont recalculées à chaque frappe.
  On ne les affiche pas d'emblée (un formulaire vide couvert de rouge est
  désagréable) : seulement pour un champ déjà quitté ("touché"), ou pour
  tous les champs après une tentative d'envoi.

  Mêmes règles que le serveur RdvMedecins (formulaire "Créer un compte") :
  titre dans la liste, prénom et nom obligatoires (20 caractères au plus),
  login en minuscules, mot de passe de 6 caractères au moins, confirmation.

  ATTENTION : cette validation n'est qu'un CONFORT pour l'utilisateur (une
  réponse immédiate, sans aller-retour avec le serveur). Elle ne protège
  rien : n'importe qui peut envoyer au serveur une requête fabriquée à la
  main (curl...). Le serveur revalide donc TOUJOURS tout (cf. exemple 20).
  =========================================================================

<script setup lang="ts">
import { computed, reactive, ref } from 'vue';

const TITRES = ['Mr', 'Mme', 'Melle'];
const FORMAT_LOGIN = /^[a-z0-9._-]{1,30}$/;
const MOT_DE_PASSE_MIN = 6;

type Champ = 'titre' | 'prenom' | 'nom' | 'login' | 'password' | 'confirmation';

const saisie = reactive<Record<Champ, string>>({
  titre: 'Mr',
  prenom: '',
  nom: '',
  login: '',
  password: '',
  confirmation: '',
});

// les champs déjà quittés par l'utilisateur (événement blur)
const touches = reactive(new Set<Champ>());
const envoiTente = ref(false);
const succes = ref(false);

// l'erreur de chaque champ ('' = pas d'erreur), recalculée à chaque frappe
const erreurs = computed<Record<Champ, string>>(() => ({
  titre: TITRES.includes(saisie.titre) ? '' : 'Choisissez un titre dans la liste.',
  prenom: obligatoire(saisie.prenom, 'Le prénom') || maximum(saisie.prenom, 'Le prénom', 20),
  nom: obligatoire(saisie.nom, 'Le nom') || maximum(saisie.nom, 'Le nom', 20),
  login:
    obligatoire(saisie.login, 'Le login') ||
    (FORMAT_LOGIN.test(saisie.login.trim())
      ? ''
      : 'Lettres minuscules non accentuées, chiffres, point, tiret, souligné.'),
  password:
    obligatoire(saisie.password, 'Le mot de passe') ||
    (saisie.password.length < MOT_DE_PASSE_MIN ? `Au moins ${MOT_DE_PASSE_MIN} caractères.` : ''),
  confirmation: saisie.confirmation === saisie.password ? '' : 'La confirmation ne correspond pas au mot de passe.',
}));

const formulaireValide = computed(() => Object.values(erreurs.value).every((e) => e === ''));

function obligatoire(valeur: string, nom: string): string {
  return valeur.trim() === '' ? `${nom} est obligatoire.` : '';
}
function maximum(valeur: string, nom: string, max: number): string {
  return valeur.trim().length > max ? `${nom} ne doit pas dépasser ${max} caractères.` : '';
}

// l'erreur à AFFICHER pour un champ : seulement s'il a été touché, ou après
// une tentative d'envoi
function erreurVisible(champ: Champ): string {
  return touches.has(champ) || envoiTente.value ? erreurs.value[champ] : '';
}

function envoyer(): void {
  envoiTente.value = true;
  succes.value = formulaireValide.value;
}
</script>

    • ligne 27 : le type des noms de champs. [Record<Champ, string>] est un objet qui a exactement ces propriétés : une faute de frappe dans un nom de champ serait détectée à la compilation ;
    • ligne 39 : les champs que l'utilisateur a déjà quittés. Un [Set] rendu réactif : [touches.add(...)] met la page à jour ;
    • lignes 44-57 : les erreurs de TOUS les champs, recalculées à chaque frappe. La chaîne vide signifie « pas d'erreur » ; [a || b] renvoie [b] quand [a] est vide : on enchaîne ainsi les contrôles d'un champ, le premier qui échoue donnant le message ;
    • ligne 59 : vrai si aucune erreur ;
    • lignes 70-72 : on n'affiche l'erreur d'un champ que s'il a été « touché », ou après une tentative d'envoi. Un formulaire vide couvert de rouge avant même qu'on ait commencé serait désagréable ;
    • ligne 85 : [novalidate] désactive les contrôles du navigateur (attribut [required]...), pour n'afficher que nos messages, dans notre style ;
    • ligne 99 : l'événement [blur] : l'utilisateur quitte le champ ;

    Une remarque essentielle : cette validation n'est qu'un confort. Elle répond immédiatement, sans aller-retour avec le serveur, mais elle ne protège rien : n'importe qui peut envoyer au serveur une requête fabriquée à la main (avec [curl], par exemple), sans passer par nos pages.
